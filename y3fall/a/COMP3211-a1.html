def build_1_layer_mlp(nbr_hidden_unit):
    mlp = nn.Sequential(
        nn.Linear(2, nbr_hidden_unit),   # W1, b1
        nn.ReLU(),
        nn.Linear(nbr_hidden_unit, 2),   # W2, b2 -> 2 class scores
    )
    return mlp

Observation (also in the notebook markdown cell)

  • Accuracy ↑ with width: h≤100: 0.875–0.900; h≥500: 0.950–0.965. Errors 20→10 for 100→500, only 8→7 for 1000→5000 ⇒ diminishing returns.
  • Boundary shape: h≤100 ⇒ nearly one straight line (≈ linear classifier), misses the diagonal red points inside the blue region; h≥1000 ⇒ visibly piecewise linear (one kink per ReLU unit), bends around that red cluster.
  • Capacity + optimisation: same budget (300 full-batch steps, lr 0.001) ⇒ wider nets fit faster; small nets underfit.
  • Limits: h=5000 still 7/200 wrong; extra width mainly costs compute; outside the data range the boundary only extrapolates linearly (meaningless).

⚠️ Exact accuracies depend on PyTorch version + CPU RNG (run: latest torch, Python 3.14, CPU); a local Anaconda run may differ in the last digits, same trend.

build_mlp

def build_mlp():
    mlp = nn.Sequential(
        nn.Linear(64, 512),   # 8x8 pixels -> 512
        nn.ReLU(),
        nn.Linear(512, 64),
        nn.ReLU(),
        nn.Linear(64, 10),    # 10 class scores
    )
    return mlp

train_mlp

def train_mlp():
    epochs = 50
    batch_size = 64
    lossfunc = torch.nn.CrossEntropyLoss()
    train_x = train_input / 16.0          # [0,16] -> [0,1]; locals, data cell untouched
    valid_x = valid_input / 16.0          # same scaling for all three splits
    test_x = test_input / 16.0

    mlp = build_mlp()
    optimizer = torch.optim.Adam(mlp.parameters(), lr=0.001)   # optimizer not fixed by task
    loss_list, valid_acc = [], []
    for i_epoch in range(epochs):
        mlp.train()
        batch_losses = []
        for batch_x, batch_y in iterate_minibatches(train_x, train_label, batch_size, shuffle=True):
            optimizer.zero_grad()
            outputs = mlp(batch_x)
            loss = lossfunc(outputs, batch_y)
            loss.backward()
            optimizer.step()              # one update per mini-batch
            batch_losses.append(loss.item())
        loss_list.append(np.mean(batch_losses))   # L_epoch = mean over B batches
        mlp.eval()
        with torch.no_grad():
            acc = AccuarcyCompute(mlp(valid_x), valid_label)   # valid acc every epoch
        valid_acc.append(acc)
        print('epoch {:2d}: train loss {:.4f}, valid acc {:.4f}'.format(i_epoch + 1, loss_list[-1], acc))

    plot_fig(loss_list, 'MNIST training loss', 'mnist_train_loss.png')      # "loss" in title -> y = loss
    plot_fig(valid_acc, 'MNIST validation accuracy', 'mnist_valid_acc.png') # else y = accuracy

    with torch.no_grad():
        test_outputs = mlp(test_x)
    test_acc = AccuarcyCompute(test_outputs, test_label)
    print('test accuracy: {:.4f}'.format(test_acc))
    predicted_label = test_outputs.argmax(dim=1).cpu().numpy()
    visualize_mnist_predictions(test_x.cpu().numpy(), test_label.cpu().numpy(), predicted_label)
