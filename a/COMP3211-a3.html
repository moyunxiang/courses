def value_iteration(transitions, gamma=0.8, tol=1e-8, max_iterations=1000):
    def q_value(values, state, action):
        return sum(p * (r + gamma * values[s2]) for p, s2, r in transitions[state][action])

    values = {s: 0.0 for s in transitions}
    iterations = 0
    while iterations < max_iterations:
        updated = {
            s: max(q_value(values, s, a) for a in transitions[s]) if transitions[s] else 0.0
            for s in transitions
        }
        iterations += 1
        delta = max(abs(updated[s] - values[s]) for s in transitions)
        values = updated
        if delta <= tol:
            break

    policy = {}
    for s in transitions:
        best_action, best_q = None, None
        for a in transitions[s]:
            q = q_value(values, s, a)
            if best_q is None or q > best_q:
                best_action, best_q = a, q
        policy[s] = best_action
    return values, policy, iterations

def epsilon_greedy(q_table, state, epsilon, rng):
    row = q_table[state]
    if rng.random() < epsilon:
        return rng.choice(list(row))
    return max(row, key=row.get)

def q_learning_update(q_table, state, action, reward, next_state, done, alpha, gamma):
    target = reward if done else reward + gamma * max(q_table[next_state].values())
    q_table[state][action] += alpha * (target - q_table[state][action])
    return q_table[state][action]

def train_q_learning(episodes=5000, alpha=0.1, gamma=0.8, epsilon=0.2, seed=0, max_steps=100):
    rng = random.Random(seed)
    env = SampleMDP(rng)
    q_table = initial_q_table()
    returns = []
    for _ in range(episodes):
        state = env.reset()
        total, discount = 0.0, 1.0
        for _ in range(max_steps):
            action = epsilon_greedy(q_table, state, epsilon, rng)
            next_state, reward, done = env.step(action)
            q_learning_update(q_table, state, action, reward, next_state, done, alpha, gamma)
            total += discount * reward
            discount *= gamma
            state = next_state
            if done:
                break
        returns.append(total)
    return q_table, returns

def legal_actions(state):
    a, b, _ = state
    return [(pile, n) for pile, size in (('A', a), ('B', b)) for n in (1, 2) if n <= size]

def next_state(state, action):
    a, b, player = state
    pile, n = action
    if pile == 'A':
        a -= n
    else:
        b -= n
    return (a, b, 'MIN' if player == 'MAX' else 'MAX')

def terminal_utility(state):
    a, b, player = state
    if a == 0 and b == 0:
        return -1 if player == 'MAX' else 1
    return None

def minimax(state):
    utility = terminal_utility(state)
    if utility is not None:
        return utility, None
    best_score, best_action = None, None
    for action in legal_actions(state):
        score, _ = minimax(next_state(state, action))
        if (best_score is None
                or (state[2] == 'MAX' and score > best_score)
                or (state[2] == 'MIN' and score < best_score)):
            best_score, best_action = score, action
    return best_score, best_action

def compute_prior(X, y):
    return np.array([np.mean(y == c) for c in (0, 1)])

def compute_likelihood(X, y):
    likelihood = np.zeros((X.shape[1], 2, 2))
    for c in (0, 1):
        for v in (0, 1):
            likelihood[:, v, c] = np.mean(X[y == c] == v, axis=0)
    return likelihood

def predict(prior, likelihood, query):
    joint = prior * np.prod(likelihood[np.arange(len(query)), query], axis=0)
    posterior = joint / joint.sum()
    return int(np.argmax(posterior)), posterior
