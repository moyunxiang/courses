def update(self, action: int, reward: float):
    self.action_counts[action] += 1                                  # N(a) += 1
    self.q_estimates[action] += (reward - self.q_estimates[action]) / self.action_counts[action]

def select_action(self, t: int = 0, max_steps=1000) -> int:
    if self.decay_type == 'linear':
        eps = self.initial_epsilon * (1 - t / max_steps)
    else:
        raise ValueError(f"Unknown decay_type: {self.decay_type}")
    eps = max(0.0, eps)                              # guard for t > T

        for a in range(env.nA):
            if pi[s, a] == 0:
                continue                                   # skip zero-prob actions
            q_sa = sum(prob * (r + gamma * V[ns]) for prob, ns, r, done in env.P[s][a])
            V_new[s] += pi[s, a] * q_sa
