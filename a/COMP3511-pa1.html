2. Ctrl-C (SIGINT)

handler only in myshell;process_cmd:SIG_DFL (and execvp resets caught signals)⇒message printed exactly once
void sigint_handler(int signum)            // signum = 2
{
    printf(TEMPLATE_MYSHELL_TERMINATE, getpid(), signum);   // same pid as start message
    exit(0);                               // flushes stdout
}

// main():
signal(SIGINT, sigint_handler);

// start of process_cmd():
signal(SIGINT, SIG_DFL);                   // children die silently on Ctrl-C
  • Terminal echoes ^C itself ⇒ do not print it; Ctrl-C → whole foreground process group.

5. Redirection (dup2)

scan segment (skip text in quotes):</>+filename→spaces⇒argv never contains them<:open(O_RDONLY)→dup2(fd, 0);>:open(O_CREAT|O_WRONLY|O_TRUNC, 0600)→dup2(fd, 1)fd 0 and fd 1 disjoint⇒< a > b≡> b < a
void extract_redirection(char *segment, char *in_file, char *out_file)
{
    int i = 0, in_quotes = 0;
    in_file[0] = '\0';
    out_file[0] = '\0';
    while (segment[i] != '\0')
    {
        char c = segment[i];
        if (c == '"')
            in_quotes = !in_quotes;
        if (!in_quotes && (c == '<' || c == '>'))
        {
            segment[i++] = ' ';                                         // blank the symbol
            take_file_name(segment, &i, c == '<' ? in_file : out_file); // copy + blank name; "<file" ok
            continue;
        }
        i++;
    }
}

void exec_segment(char *segment)          // never returns
{
    ...
    extract_redirection(segment, in_file, out_file);
    if (in_file[0] != '\0')
    {
        fd = open(in_file, O_RDONLY);
        if (fd < 0) exit(1);
        dup2(fd, STDIN_FILENO);           // atomically close 0, make 0 a copy of fd
        close(fd);                        // no leaked fd in the exec'd program
    }
    if (out_file[0] != '\0')
    {
        fd = open(out_file, O_CREAT | O_WRONLY | O_TRUNC, S_IRUSR | S_IWUSR);  // 0600
        if (fd < 0) exit(1);
        dup2(fd, STDOUT_FILENO);
        close(fd);
    }
    parse_arguments(arguments, segment, &num_arguments, SPACE_CHARS);
    if (num_arguments == 0) exit(0);
    arguments[num_arguments] = NULL;      // <= 8 args -> index <= 8 < 9 slots
    execvp(arguments[0], arguments);
    exit(1);                              // reached only if execvp fails
}
  • Mode 0600 = lab example = reference. Reference has no O_TRUNC (spec: file does not exist); added so an existing file is overwritten cleanly.

6. Splitting pipe segments

parse_arguments(..., "|") strips quotes⇒"Hello Hong Kong"→3 args ×;manual split outside quotes⇒2 args ✓
int split_pipe_segments(char *cmdline, char **segments)
{
    int n = 0, in_quotes = 0;
    char *p = cmdline;
    segments[n++] = cmdline;
    for (; *p != '\0'; p++)
    {
        if (*p == '"')
            in_quotes = !in_quotes;
        else if (!in_quotes && *p == PIPE_CHAR[0] && n < MAX_PIPE_SEGMENTS)  // n <= 8 slots
        {
            *p = '\0';                    // segments are substrings of cmdline, no copies
            segments[n++] = p + 1;
        }
    }
    return n;
}

7. Multi-level pipe

child i:stdin←pipei−1[0] (i>0),stdout→pipei[1] (i<n−1)invariant:at start of iteration i+1, parent holds only pipei[0]⇒open write ends of pipei={child i’s stdout}⇒child i exits⇒child i+1 gets EOF
if (num_pipe_segments == 1)
    exec_segment(pipe_segments[0]);       // no pipe: exec directly, no extra process

for (i = 0; i < num_pipe_segments; i++)
{
    int is_last = (i == num_pipe_segments - 1);
    if (!is_last)
        pipe(pfds);                       // pipe_i before fork_i: child i inherits it

    pid_t pid = fork();
    if (pid == 0)
    {
        if (prev_read != -1) { dup2(prev_read, STDIN_FILENO); close(prev_read); }
        if (!is_last) { close(pfds[0]); dup2(pfds[1], STDOUT_FILENO); close(pfds[1]); }
        exec_segment(pipe_segments[i]);   // child keeps only fds 0,1,2; file redirs override pipe
    }
    if (prev_read != -1)
        close(prev_read);                 // only child i reads pipe_{i-1}
    if (!is_last) { close(pfds[1]); prev_read = pfds[0]; }   // only child i writes pipe_i
}
for (i = 0; i < num_pipe_segments; i++)
    wait(0);                              // reap all n children: no zombies
exit(0);                                  // main's wait(0) returns -> next prompt after whole pipeline
Property Reason
EOF reaches sort / wc parent closes pfds[1] after each fork (else readers block forever)
No deadlock all n children forked before any wait; waiting inside loop ⇒ writer blocks on full 64 KB buffer with reader not yet forked
Hidden01 count single command ⇒ forked child execvps directly ⇒ 3 myshell in ps
Extra process during a pipe command: +1 process_cmd (blocked in wait), exits after pipeline; reference same

Platform differences

  • ⚠️ macOS ps: all terminals of this user, no ps line; Linux: current terminal only (tcsh + 3× myshell + ps, as in spec). Criterion same: 3 myshell.
  • ⚠️ macOS wc -l pads ( 429); GNU wc on stdin prints 429; ls -lh date format and @ xattr marker differ — external commands, not myshell.
  • ⚠️ signal(): BSD semantics on macOS; glibc with _GNU_SOURCE (skeleton defines it) also BSD (SA_RESTART, no reset); handler calls exit ⇒ same behaviour.
  • ⚠️ No gcc 11 / glibc here; skeleton redefines STDIN_FILENO with the same value as <unistd.h> (no warning here). Recheck 0 warnings with gcc -pedantic -Wall -Wextra -std=c99 on Lab 2.
